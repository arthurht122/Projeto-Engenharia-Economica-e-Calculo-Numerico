# k = ganho extra do juros composto
def calcula_k(i, n):
    return (1 + i)**n - (n*i + 1)

k = calcula_k(0.05, 12)
print(f"Lucro extra: {k*100:.2f}%")
# > Lucro extra: 19.59%
